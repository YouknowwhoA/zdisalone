import numpy as np
import matplotlib.pyplot as plt

# =========================
# Parameters (match your notes)
# =========================
v_star = 20.0
I_K = 30.0
I_Na = 150.0
tau = 10.0
C = 1.0
g = 1.0

# This is just a display scaling for w on the y-axis:
# w = W_SCALE * theta. Adjust if your y-axis numbers differ.
W_SCALE = 35.0

# Theta sweep range (should cover the branches in your plot)
THETA_MIN = 0.20
THETA_MAX = 4.00
N_THETA = 260

# =========================
# Small utilities
# =========================
def exp_clip(z):
    # avoid overflow/underflow
    return np.exp(np.clip(z, -700.0, 700.0))

def safe_log(x):
    if x <= 0:
        return np.nan
    return np.log(x)

# =========================
# Exact traveling wave machinery (using your page-7 formulas)
# =========================
def a_from_theta(theta):
    # Using definition of theta in your notes: a = r/(2 rho w^2) = C^2/(2 g theta^2)
    return C**2 / (2.0 * g * theta**2)

def lambdas(theta):
    a = a_from_theta(theta)
    disc = C**2 + 4.0*a*g
    s = np.sqrt(disc)
    lam_plus = (C + s) / (2.0*a)
    # stable lam_minus (avoids cancellation):
    lam_minus = -2.0*g / (C + s)
    return lam_minus, lam_plus

def Delta(theta):
    a = a_from_theta(theta)
    return g - C/tau - a/(tau**2)

def T0T1_from_N(N0, N1):
    # N0, N1 must satisfy 0 < N0 < N1 < 1
    if not (0.0 < N0 < N1 < 1.0):
        return np.nan, np.nan
    T0 = tau * safe_log(N1 / N0)
    T1 = tau * safe_log((1.0 - N0) / (1.0 - N1))
    return T0, T1

def coeffs_and_residuals(I0, theta, N0, N1):
    """
    Build b1,a1,b2,a2 from your closed-form formulas,
    then return residuals:
      F1 = V1'(0) - V2'(0)
      F2 = V1'(-T0) - V2'(T1)
    """
    T0, T1 = T0T1_from_N(N0, N1)
    if not np.isfinite(T0) or not np.isfinite(T1):
        return None

    D = Delta(theta)
    if not np.isfinite(D) or abs(D) < 1e-10:
        return None

    lam_minus, lam_plus = lambdas(theta)

    # exponent helpers
    em_lp_T0 = exp_clip(-lam_plus * T0)
    em_lp_T1 = exp_clip(-lam_plus * T1)
    e_lm_T0 = exp_clip(lam_minus * T0)  # lam_minus < 0
    e_lm_T1 = exp_clip(lam_minus * T1)

    # Particular solutions:
    # interval 1: M=0, N(T)=N0*exp(-T/tau) on [-T0,0]
    A1 = I0 / g
    B1 = -(I_K * N0) / D
    # Vp1(T)=A1 + B1 exp(-T/tau)

    # interval 2: M=1, N(T)=1-(1-N0)exp(-T/tau) on [0,T1]
    A2 = (I0 - I_K) / g
    B2 = ((I_Na + I_K) * (1.0 - N0)) / D
    # Vp2(T)=A2 + B2 exp(-T/tau)

    # Needed exp(-T/tau) at endpoints
    e_T0_over_tau = exp_clip(T0 / tau)               # exp(-(-T0)/tau)
    e_mT1_over_tau = exp_clip(-T1 / tau)             # exp(-T1/tau)

    # Vp values
    Vp1_mT0 = A1 + B1 * e_T0_over_tau
    Vp1_0   = A1 + B1
    Vp2_0   = A2 + B2
    Vp2_T1  = A2 + B2 * e_mT1_over_tau

    # Vp derivatives
    Vp1p_mT0 = -(B1/tau) * e_T0_over_tau
    Vp1p_0   = -(B1/tau)
    Vp2p_0   = -(B2/tau)
    Vp2p_T1  = -(B2/tau) * e_mT1_over_tau

    # Right-hand side constants for boundary conditions V=v*
    rhs1_mT0 = v_star - Vp1_mT0
    rhs1_0   = v_star - Vp1_0

    rhs2_0   = v_star - Vp2_0
    rhs2_T1  = v_star - Vp2_T1

    # b1, a1 (your page-7 formulas)
    den1 = 1.0 - exp_clip((lam_minus - lam_plus) * T0)
    if abs(den1) < 1e-14:
        return None
    b1 = (rhs1_0 - rhs1_mT0 * e_lm_T0) / den1
    a1 = rhs1_mT0 - b1 * em_lp_T0

    # b2, a2
    den2 = 1.0 - exp_clip((lam_minus - lam_plus) * T1)
    if abs(den2) < 1e-14:
        return None
    b2 = (rhs2_T1 - rhs2_0 * e_lm_T1) / den2
    a2 = rhs2_0 - b2 * em_lp_T1

    # Derivatives at matching points
    # V1'(0), V2'(0)
    V1p_0 = Vp1p_0 + a1 * lam_minus * e_lm_T0 + b1 * lam_plus
    V2p_0 = Vp2p_0 + a2 * lam_minus          + b2 * lam_plus * em_lp_T1

    # V1'(-T0), V2'(T1)
    V1p_mT0 = Vp1p_mT0 + a1 * lam_minus + b1 * lam_plus * em_lp_T0
    V2p_T1  = Vp2p_T1  + a2 * lam_minus * e_lm_T1 + b2 * lam_plus

    F1 = V1p_0  - V2p_0
    F2 = V1p_mT0 - V2p_T1

    out = {
        "T0": T0, "T1": T1, "P": T0+T1,
        "a1": a1, "b1": b1, "a2": a2, "b2": b2,
        "lam_minus": lam_minus, "lam_plus": lam_plus,
        "A1": A1, "B1": B1, "A2": A2, "B2": B2,
        "D": D,
        "F": np.array([F1, F2], dtype=float),
    }
    return out

def newton_solve_N0N1(I0, theta, x0, max_iter=40, tol=1e-10):
    """
    Pure numpy Newton with finite-difference Jacobian + backtracking.
    x = [N0, N1]
    """
    x = np.array(x0, dtype=float)

    def project(x):
        N0, N1 = x
        N0 = float(np.clip(N0, 1e-8, 1.0-1e-8))
        N1 = float(np.clip(N1, N0 + 1e-6, 1.0-1e-8))
        return np.array([N0, N1], dtype=float)

    x = project(x)

    info = coeffs_and_residuals(I0, theta, x[0], x[1])
    if info is None:
        return None
    F = info["F"]
    nF = np.linalg.norm(F)

    for _ in range(max_iter):
        if nF < tol:
            return x

        # finite-difference Jacobian
        h = 1e-6
        J = np.zeros((2,2), dtype=float)
        for j in range(2):
            dx = np.zeros(2)
            dx[j] = h
            xp = project(x + dx)
            xm = project(x - dx)

            infop = coeffs_and_residuals(I0, theta, xp[0], xp[1])
            infom = coeffs_and_residuals(I0, theta, xm[0], xm[1])
            if (infop is None) or (infom is None):
                return None
            J[:, j] = (infop["F"] - infom["F"]) / (xp[j] - xm[j] + 1e-18)

        # solve for step
        try:
            step = np.linalg.solve(J, F)
        except np.linalg.LinAlgError:
            return None

        # backtracking line search
        alpha = 1.0
        accepted = False
        for _ls in range(12):
            x_new = project(x - alpha * step)
            info_new = coeffs_and_residuals(I0, theta, x_new[0], x_new[1])
            if info_new is not None:
                F_new = info_new["F"]
                nF_new = np.linalg.norm(F_new)
                if nF_new < (1.0 - 1e-4*alpha) * nF:
                    x, F, nF = x_new, F_new, nF_new
                    accepted = True
                    break
            alpha *= 0.5

        if not accepted:
            return None

    return None

# =========================
# epsilon -> 0 approximation curve (param by theta)
# =========================
def gamma_from_theta(theta):
    s = np.sqrt(1.0 + 2.0*theta**2)
    return (s + 1.0) / (s - 1.0)

def period_approx_from_theta(I0, theta):
    """
    From your page-10 formula:
      E = exp(P/tau) = (INa*gamma + A*(gamma+1)) / (INa + A*(gamma+1))
      A = I0 - g*v*
      P = tau * ln(E)
    """
    gam = gamma_from_theta(theta)
    A = I0 - g*v_star
    denom = (I_Na + A*(gam + 1.0))
    num   = (I_Na*gam + A*(gam + 1.0))
    if denom == 0:
        return np.nan
    E = num / denom
    if E <= 1.0:  # period would be <=0 or invalid
        return np.nan
    return tau * np.log(E)

# =========================
# Build exact branch for each I0
# =========================
def compute_exact_branch(I0, theta_grid):
    seeds = [
        (0.20, 0.80),
        (0.30, 0.85),
        (0.10, 0.70),
        (0.05, 0.95),
        (0.40, 0.90),
    ]
    P_list, w_list, theta_list, sol_list = [], [], [], []
    guess = None

    for theta in theta_grid:
        x = None
        if guess is not None:
            x = newton_solve_N0N1(I0, theta, guess)
        if x is None:
            for s in seeds:
                x = newton_solve_N0N1(I0, theta, s)
                if x is not None:
                    break
        if x is None:
            guess = None
            continue

        info = coeffs_and_residuals(I0, theta, x[0], x[1])
        if info is None or not np.isfinite(info["P"]):
            guess = None
            continue

        P_list.append(info["P"])
        w_list.append(W_SCALE * theta)
        theta_list.append(theta)
        sol_list.append(x)
        guess = x

    return np.array(P_list), np.array(w_list), np.array(theta_list), sol_list

# =========================
# Wave profile builder (exact)
# =========================
def build_wave_profile(I0, theta, N0N1):
    N0, N1 = N0N1
    info = coeffs_and_residuals(I0, theta, N0, N1)
    if info is None:
        return None

    T0, T1, P = info["T0"], info["T1"], info["P"]
    a1, b1, a2, b2 = info["a1"], info["b1"], info["a2"], info["b2"]
    lam_minus, lam_plus = info["lam_minus"], info["lam_plus"]
    A1, B1, A2, B2 = info["A1"], info["B1"], info["A2"], info["B2"]

    def Vp1(t):
        return A1 + B1 * exp_clip(-t/tau)

    def Vp2(t):
        return A2 + B2 * exp_clip(-t/tau)

    def V1(t):
        # t in [-T0,0]
        return Vp1(t) + a1 * exp_clip(lam_minus*(t + T0)) + b1 * exp_clip(lam_plus*t)

    def V2(t):
        # t in [0,T1]
        return Vp2(t) + a2 * exp_clip(lam_minus*t) + b2 * exp_clip(lam_plus*(t - T1))

    t1 = np.linspace(-T0, 0.0, 400)
    t2 = np.linspace(0.0, T1, 400)

    V1_vals = np.array([V1(t) for t in t1])
    V2_vals = np.array([V2(t) for t in t2])

    # shifted copy of V1 to show periodicity
    t1_shift = t1 + P
    V1_shift_vals = V1_vals.copy()

    return {
        "t1": t1, "V1": V1_vals,
        "t2": t2, "V2": V2_vals,
        "t1_shift": t1_shift, "V1_shift": V1_shift_vals,
        "T0": T0, "T1": T1, "P": P
    }

    if len(P_exact) > 0:
        plt.plot(P_exact, w_exact, linewidth=2, label=f"Exact, I0={I0:g}")
    # approx (param by theta)
    P_app = []
    w_app = []
    for th in np.linspace(THETA_MAX, THETA_MIN, N_THETA):
        Pth = period_approx_from_theta(I0, th)
        if np.isfinite(Pth):
            P_app.append(Pth)
            w_app.append(W_SCALE * th)
    if len(P_app) > 0:
        plt.plot(P_app, w_app, "--", linewidth=2, label=f"$\\varepsilon\\to0$, I0={I0:g}")
